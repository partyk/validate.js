validate.js

Vanilla JS · jQuery Validation syntax · zero dependencies · i18n

✓ Form submitted successfully!
Usage
// Load locale (optional)
validate.setLocale('cs');

// Initialize — same syntax as jQuery Validation
validate('#myForm', {
  rules: {
    username: { required: true, minlength: 3 },
    email:    { required: true, email: true },
    password: { required: true, minlength: 6 },
    terms:    { required: true },
  },
  submitHandler(form) {
    // called only when form is valid
    form.submit();
  }
});
